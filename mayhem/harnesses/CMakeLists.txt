cmake_minimum_required(VERSION 3.20)
project(rclcpp_fuzz C CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(rclcpp REQUIRED)
find_package(rcl_interfaces REQUIRED)
find_package(rcl_yaml_param_parser REQUIRED)
find_package(fastcdr REQUIRED)
find_package(rclcpp_components REQUIRED)
find_package(composition_interfaces REQUIRED)

add_library(standalone_main OBJECT ${STANDALONE_MAIN})
set_source_files_properties(${STANDALONE_MAIN} PROPERTIES LANGUAGE C)

foreach(h fuzz_param_service fuzz_param_yaml fuzz_ros_args fuzz_param_event fuzz_names)
  add_executable(${h} ${h}.cpp ${LSAN_OFF_OBJ})
  target_link_libraries(${h} rclcpp::rclcpp ${rcl_interfaces_TARGETS} rcl_yaml_param_parser::rcl_yaml_param_parser fastcdr)
  target_link_options(${h} PRIVATE ${FUZZ_ENGINE})
  target_compile_options(${h} PRIVATE -fsanitize=fuzzer-no-link)
  add_executable(${h}-standalone ${h}.cpp ${LSAN_OFF_OBJ} $<TARGET_OBJECTS:standalone_main>)
  target_link_libraries(${h}-standalone rclcpp::rclcpp ${rcl_interfaces_TARGETS} rcl_yaml_param_parser::rcl_yaml_param_parser fastcdr)
endforeach()

add_executable(fuzz_component_manager fuzz_component_manager.cpp ${LSAN_OFF_OBJ})
add_executable(fuzz_component_manager-standalone fuzz_component_manager.cpp ${LSAN_OFF_OBJ} $<TARGET_OBJECTS:standalone_main>)
foreach(h fuzz_component_manager fuzz_component_manager-standalone)
  target_link_libraries(${h} rclcpp_components::component_manager rclcpp::rclcpp ${composition_interfaces_TARGETS} fastcdr)
endforeach()
target_link_options(fuzz_component_manager PRIVATE ${FUZZ_ENGINE})
target_compile_options(fuzz_component_manager PRIVATE -fsanitize=fuzzer-no-link)
